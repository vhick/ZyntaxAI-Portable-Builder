cmake_minimum_required(VERSION 3.20)
project(ZyntaxAIPortable LANGUAGES CXX RC)
add_executable(ZyntaxAIPortable WIN32 ZyntaxAIPortable.cpp icon.rc)
target_compile_features(ZyntaxAIPortable PRIVATE cxx_std_17)
target_compile_definitions(ZyntaxAIPortable PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
  set_property(TARGET ZyntaxAIPortable PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(ZyntaxAIPortable PRIVATE /W4 /EHsc /O2)
endif()
target_link_libraries(ZyntaxAIPortable PRIVATE shell32)
